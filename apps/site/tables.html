CREATE TYPE house_status AS ENUM ('viewing', 'offer', 'closed');

CREATE TABLE house (
    id         uuid PRIMARY KEY DEFAULT uuidv7(),
    space      uuid NOT NULL REFERENCES archive.space,
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),
    address    text NOT NULL,
    price      archive.money,
    status     house_status NOT NULL DEFAULT 'viewing',
    contact    archive.phone,
    owner      uuid REFERENCES person,
    search     tsvector GENERATED ALWAYS AS (archive.search_doc(address)) STORED
);
CREATE INDEX house_search ON house USING gin (search);

Questions.

More in the tables docs.

Do I need a database expert to change things?

No. Make your change, and Archive works out the steps and shows you the plan before anything happens.

Can I delete a list by accident?

No. Deleting a list takes a deliberate extra step. Boards and calendars hold no records of their own, so removing one is always safe.

If I rename a field, do I lose what is in it?

No. Mark it as a rename and everything in the field comes along, as part of the same change.

Can my developers work with the data directly?

Yes. The same rules apply, so they see exactly what the app shows you and nothing more.

Who can change the structure?

Admins of the workspace or of a space. Everyone else adds and edits records in the spaces they belong to.
