const [hit] = await archive.search("Northwind master agreement", { type: "file", limit: 1 })
const doc = await archive.read(hit.id)
const terms = doc.paragraphs.find((p) => p.text.includes("Net 60"))

await archive.edit(hit.id, [
  { op: "replace_text", para: terms.id, find: "Net 60", replace: "Net 30" },
  { op: "insert_paragraph_after", para: terms.id, text: "Late invoices accrue 1.5% per month." },
  { op: "set_cell_text", table: 0, row: 1, col: 2, text: "$48,000" },
])

return archive.read(hit.id)
const budget = await archive.create("Budget.xlsx")

await archive.edit(budget.id, [
  { op: "set_cell", cell: "A1", value: "Item" },
  { op: "set_cell", cell: "B1", value: "Cost" },
  { op: "set_cell", cell: "A2", value: "Hosting" },
  { op: "set_cell", cell: "B2", value: 1200 },
  { op: "set_cell", cell: "A3", value: "Design" },
  { op: "set_cell", cell: "B3", value: 4800 },
  { op: "set_cell", cell: "A4", value: "Total" },
  { op: "set_formula", cell: "B4", formula: "SUM(B2:B3)" },
])

return archive.read(budget.id)

Questions.

More in the files docs.

Will my formatting survive an edit?

Yes. New text picks up the formatting of the text it replaces, and the rest of the file stays exactly as it was.

Do formulas recalculate?

Yes. Every formula is worked out fresh the moment the workbook is opened.

What can my AI not change yet?

Adding or removing lines in a sheet, whole slides or shapes, and formatting such as bold or number styles. It can always see exactly what it is able to edit.

How large can a file be?

Up to 64 MB in a single upload. Bigger files go up in pieces and resume if the connection drops.

What happens when two assistants edit the same file?

They take turns, and each edit goes on top of the latest version, so nothing gets lost.
