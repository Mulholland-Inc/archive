How agents work
const hits = await archive.search("Northwind terms", {
  type: "file",
})

for (const hit of hits.slice(0, 5)) {
  await archive.create("contract", {
    party: "Northwind",
    terms: "30 days",
    source: hit.id,
  })
}

return archive.list("contract", {
  where: { party: "Northwind" },
})
